export const api = {
  scanDevices: () => fetch(`${BASE_URL}/devices/scan`).then(r => r.json()),
  connectDevice: (comPort: string, serial: string) => 
    fetch(`${BASE_URL}/devices/connect?com_port=${comPort}&serial_number=${serial}`, {method: 'POST'}),
  getConnectedDevices: () => fetch(`${BASE_URL}/devices/connected`).then(r => r.json()),
  uploadScript: (serial: string, script: string) =>
    fetch(`${BASE_URL}/devices/upload-script/${serial}`, {
      method: 'POST',
      headers: {'Content-Type': 'application/json'},
      body: JSON.stringify({script})
    }),
  runScript: (serial: string, metadata: string) =>
    fetch(`${BASE_URL}/devices/run-script/${serial}?metadata=${metadata}`, {method: 'POST'}),
  disconnectDevice: (serial: string) =>
    fetch(`${BASE_URL}/devices/disconnect/${serial}`, {method: 'POST'})
};

Example WebSocket Hook
typescriptfunction useWebSocket(serialNumber: string) {
  const [events, setEvents] = useState<WebSocketEvent[]>([]);
  
  useEffect(() => {
    const ws = new WebSocket(`ws://localhost:8000/ws/device/${serialNumber}`);
    
    ws.onmessage = (event) => {
      const data = JSON.parse(event.data);
      setEvents(prev => [...prev, data]);
    };
    
    return () => ws.close();
  }, [serialNumber]);
  
  return events;
}
